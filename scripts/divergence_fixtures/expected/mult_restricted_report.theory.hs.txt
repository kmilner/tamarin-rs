theory MultRestrictedReport

begin

// Function signature and definition of the equational theory E

builtins: diffie-hellman
functions: fst/1, pair/2, snd/1
equations: fst(<x.1, x.2>) = x.1, snd(<x.1, x.2>) = x.2

rule (modulo E) ConclusionProduct:
   [ Fr( ~a ), Fr( ~b ), Fr( ~c ) ]
  --[ ActionOnly( (~a*~c) ), ActionOnly( (~a*~c*~c) ), ActionOnly( ~b ) ]->
   [ Out( <(~a*~b), (~a*~a*~b)> ) ]

rule (modulo E) LhsAbstraction:
   [ In( <fst(z), snd(w)> ) ] --[ Seen( 'c' ) ]-> [ Out( <z, w> ) ]



end
