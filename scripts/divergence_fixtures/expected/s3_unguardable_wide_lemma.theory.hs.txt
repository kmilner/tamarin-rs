lemma WideUnguardable:
  all-traces
  "∀ firstArgument secondArgument thirdArgument fourthArgument.
    ((<firstArgument, secondArgument> = <thirdArgument, fourthArgument>) ∧
     (<secondArgument, thirdArgument> = <fourthArgument, firstArgument>)) ⇒
    (⊥)"
/*
conversion to guarded formula failed:
  unguarded variable(s) 'firstArgument', 'secondArgument', 'thirdArgument',
  'fourthArgument' in the subformula
    "∀ firstArgument secondArgument thirdArgument fourthArgument.
      ((<firstArgument, secondArgument> = <thirdArgument, fourthArgument>) ∧
       (<secondArgument, thirdArgument> = <fourthArgument, firstArgument>)) ⇒
      (⊥)"
  in the formula
    "∀ firstArgument secondArgument thirdArgument fourthArgument.
      ((<firstArgument, secondArgument> = <thirdArgument, fourthArgument>) ∧

lemma NarrowUnguardable:
