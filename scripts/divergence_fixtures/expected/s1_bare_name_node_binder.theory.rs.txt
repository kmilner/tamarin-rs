theory BareNameNodeBinder

begin

// Function signature and definition of the equational theory E

functions: fst/1, pair/2, snd/1
equations: fst(<x.1, x.2>) = x.1, snd(<x.1, x.2>) = x.2

rule (modulo E) A:
   [ Fr( ~x ) ] --[ Alive( ~x ), Once( ~x ) ]-> [ ]

  /* has exactly the trivial AC variant */

lemma eq_rhs:
  all-traces
  "∀ y z #k #l. ((Alive( y ) @ #k) ∧ (Alive( z ) @ #l)) ⇒ (#k = #l)"
/*
guarded formula characterizing all counter-examples:
"∃ y z #k #l.1. (Alive( y ) @ #k) ∧ (Alive( z ) @ #l.1) ∧ ¬(#k = l)"
*/
by sorry

lemma eq_rhs_suffix_lhs:
  all-traces
  "∀ y z #k #l. ((Alive( y ) @ #k) ∧ (Alive( z ) @ #l)) ⇒ (#k = #l)"
/*
guarded formula characterizing all counter-examples:
"∃ y z #k #l.1. (Alive( y ) @ #k) ∧ (Alive( z ) @ #l.1) ∧ ¬(#k = l)"
*/
by sorry

lemma eq_rhs_indexed:
  all-traces
  "∀ y z #k #l. ((Alive( y ) @ #k) ∧ (Alive( z ) @ #l)) ⇒ (#k = #l)"
/*
guarded formula characterizing all counter-examples:
"∃ y z #k #l.2. (Alive( y ) @ #k) ∧ (Alive( z ) @ #l.2) ∧ ¬(#k = l.1)"
*/
by sorry

lemma eq_both_binders:
  all-traces "∀ l #l.1. (Alive( l ) @ #l.1) ⇒ (#l.1 = l)"
/*
guarded formula characterizing all counter-examples:
"∃ l #l.1. (Alive( l ) @ #l.1) ∧ ¬(#l.1 = l)"
*/
by sorry

lemma fact_arg:
  all-traces "∀ #l. (Alive( #l ) @ #l) ⇒ (⊥)"
/*
guarded formula characterizing all counter-examples:
"∃ #l.1. (Alive( l ) @ #l.1)"
*/
by sorry

lemma disjunction:
  exists-trace
  "∃ #j1 #l1.
    (((Once( 'a' ) @ #j1) ∧ (Once( 'b' ) @ #l1)) ∧ (#j1 < #l1)) ∧
    (∀ a #k. (Once( a ) @ #k) ⇒ ((#k = #j1) ∨ (#k = #l1)))"
/*
guarded formula characterizing all satisfying traces:
"∃ #j1 #l1.1.
  (Once( 'a' ) @ #j1) ∧ (Once( 'b' ) @ #l1.1)
 ∧
  (#j1 < #l1.1) ∧ (∀ a #k. (Once( a ) @ #k) ⇒ ((#k = #j1) ∨ (#k = l1)))"
*/
by sorry



end
