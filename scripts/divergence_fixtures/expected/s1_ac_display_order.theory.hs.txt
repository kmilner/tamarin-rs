theory AcDisplayOrder

begin

// Function signature and definition of the equational theory E

builtins: multiset
functions: fst/1, pair/2, snd/1
equations: fst(<x.1, x.2>) = x.1, snd(<x.1, x.2>) = x.2

rule (modulo E) A:
   [ Fr( ~x ), In( y ) ] --[ B( (~x++y) ), C( y ) ]-> [ ]

  /* has exactly the trivial AC variant */

lemma nested_same_name:
  all-traces "∀ x #i. (C( x ) @ #i) ⇒ (∃ ~x.1 #j. B( (x++~x.1) ) @ #j)"
/*
guarded formula characterizing all counter-examples:
"∃ x #i. (C( x ) @ #i) ∧ ∀ ~x.1 #j. (B( (x++~x.1) ) @ #j) ⇒ ⊥"
*/
by sorry

lemma nested_same_name_written_sorted:

*/
by sorry



end
